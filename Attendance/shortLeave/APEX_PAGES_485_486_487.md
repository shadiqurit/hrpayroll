SELECT 
    r.id,
    r.req_no,
    r.req_category_desc                                AS category,
    r.purpose_type_desc                                AS purpose_type,
    r.req_date_char                                    AS leave_date,
    r.out_time_char || ' - ' || r.in_time_char         AS timings,

    r.reason_purpose                                   AS reason,
    r.destination_place                                AS destination,
    r.status_display_text                              AS status,
    r.status                                           AS raw_status,
    -- Edit/View Link
    CASE 
        WHEN r.status = 'D' THEN
            APEX_PAGE.GET_URL(
                p_page        => '486',
                p_clear_cache => '486',
                p_items       => 'P486_ID',
                p_values      => r.id

        ELSE
            APEX_PAGE.GET_URL(
                p_page        => '486',
                p_clear_cache => '486',
                p_items       => 'P486_ID,P486_READONLY',
                p_values      => r.id || ',Y'

FROM HRMS.v_short_leave_my_report r
WHERE r.empid = :USER_ID
ORDER BY r.req_date DESC, r.id DESC;

- **False Action:** Hide -> (উপরে উল্লেখিত আইটেমগুলো)।

#### 2. DA_CHECK_MONTHLY_QUOTA (লাইভ মাসিক কোটা সতর্কতা)
- **Event:** Change
- **Selection Type:** Items -> `P486_REQ_DATE`, `P486_REQ_CATEGORY`, `P486_PURPOSE_TYPE`
- **Action:** Set Value
- **Set Type:** PL/SQL Function Body
- **Page Items to Submit:** `P486_REQ_DATE,P486_REQ_CATEGORY,P486_PURPOSE_TYPE,P486_ID`
- **PL/SQL Code:**
```sql
DECLARE
    l_used NUMBER;
    l_date DATE;
BEGIN
    BEGIN l_date := TO_DATE(:P486_REQ_DATE, 'YYYY-MM-DD'); EXCEPTION WHEN OTHERS THEN l_date := SYSDATE; END;

    IF :P486_REQ_CATEGORY = 'SHORT_LEAVE' AND :P486_PURPOSE_TYPE = 'PERSONAL' THEN
        l_used := HRMS.fn_get_monthly_sl_count(
            p_empid        => :USER_ID,
            p_req_date     => l_date,
            p_req_category => 'SHORT_LEAVE',
            p_purpose_type => 'PERSONAL',
            p_exclude_id   => :P486_ID
        );
        IF l_used >= 3 THEN
            RETURN '<div style="color:red; font-weight:bold; padding:4px; border:1px solid red; border-radius:4px;">' ||
                   '⚠️ সতর্কতা: চলতি মাসে আপনি ইতোমধ্যে ৩টি শর্ট লিভ নিয়েছেন/আবেদন করেছেন। আর আবেদন করা যাবে না।</div>';
        ELSE
            RETURN '<div style="color:green; font-weight:bold; padding:4px;">' ||
                   'ℹ️ চলতি মাসে ব্যবহৃত: ' || l_used || ' / ৩ টি। অবশিষ্ট আছে: ' || (3 - l_used) || ' টি।</div>';
        END IF;
    ELSE
        RETURN '<div style="color:#555; padding:4px;">অফিসিয়াল কাজের জন্য মাসিক ৩টির কোটা প্রযোজ্য নয়।</div>';
    END IF;
END;
```
